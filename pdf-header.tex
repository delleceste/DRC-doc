% Preamble for building NOTES.pdf with pandoc + pdflatex (no xelatex here).
% pdflatex has no native Unicode, so every non-ASCII glyph used in NOTES.md is
% mapped explicitly below.  Regenerate the list with:
%   python3 -c "import collections;s=open('NOTES.md').read();print(sorted({c for c in s if ord(c)>127}))"
\usepackage{amsmath,amssymb}
\usepackage{newunicodechar}
\usepackage{etoolbox}
\usepackage{booktabs,longtable}

% --- long file paths in typewriter must be breakable or they run off the page
%     (e.g. ../DRC-120.blue/120-blue-with-inversion.mdat overflowed by 111pt) ---
\usepackage[htt]{hyphenat}
\setlength{\emergencystretch}{3em}

% --- wide tables: shrink so the frequency tables fit the text block ---
\AtBeginEnvironment{longtable}{\footnotesize}
\setlength{\tabcolsep}{3.5pt}
\renewcommand{\arraystretch}{1.05}

% --- figures: never wider than the text block ---
% graphicx is loaded here explicitly: pandoc only pulls it in when the document
% contains an image, so \setkeys{Gin} fails on documents that have none.
\usepackage{graphicx}
\setkeys{Gin}{width=\linewidth,keepaspectratio}
